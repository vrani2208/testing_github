QUERY 1 - SELECT and WHERE
==========================

SELECT *
FROM books
WHERE rating >= 4;


QUERY 2 - ORDER BY
==================

SELECT title, price_gbp
FROM books
ORDER BY price_gbp DESC;


QUERY 3 - LIMIT
===============

SELECT title, price_gbp
FROM books
ORDER BY price_gbp DESC
LIMIT 10;


QUERY 4 - DISTINCT
==================

SELECT DISTINCT rating
FROM books;


QUERY 5 - BETWEEN and IN
========================

SELECT *
FROM books
WHERE price_gbp BETWEEN 20 AND 40
AND rating IN (4, 5);


QUERY 6 - JOIN
==============

SELECT
    b.title,
    b.price_gbp,
    c.category_name
FROM books b
JOIN categories c
    ON b.category_id = c.category_id;
